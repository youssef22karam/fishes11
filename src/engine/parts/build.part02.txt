  // ── EYES ──────────────────────────────────────────────────────────────────
  if (sp.eye.d > 0) {
    const e = sp.eye;
    const r = (e.d / 2) * L;
    const sides: { side: 1 | -1; s: number; y: number }[] = [];
    if (sp.plan === 'flatR' || sp.plan === 'flatL') {
      const sd = sp.plan === 'flatR' ? 1 : -1;
      sides.push({ side: sd, s: e.s, y: e.y }, { side: sd, s: e.s + e.d * 0.9, y: e.y - 0.38 });
    } else sides.push({ side: 1, s: e.s, y: e.y }, { side: -1, s: e.s, y: e.y });
    const irisB = new Batch(), pupilB = new Batch(), corneaB = new Batch(), ringB = new Batch();
    const sph = new THREE.SphereGeometry(1, 32, 20);
    const tor = new THREE.TorusGeometry(1, 0.14, 10, 32);
    for (const { side, s, y } of sides) {
      const at = model.attach(s, y, side);
      const q = new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0, 0, 1), at.n);
      const ctr = at.p.clone().addScaledVector(at.n, -r * 0.42);
      const pl = model.plasticAt(s, at.phi, false);
      const eyePl: [number, number, number, number] = [pl[0] * 0.3, pl[1] * 0.5, 0, 0];
      const M = (off: number, sx: number, sy: number, sz: number) =>
        new THREE.Matrix4().compose(ctr.clone().addScaledVector(at.n, off), q, new THREE.Vector3(sx, sy, sz));
      irisB.add(sph, M(0, r, r, r * 0.92), 0, undefined, 0, eyePl);
      pupilB.add(sph, M(r * 0.62, r * 0.5, r * 0.5, r * 0.28), 0, undefined, 0, eyePl);
      corneaB.add(sph, M(r * 0.05, r * 1.03, r * 1.03, r * 1.0), 0, undefined, 0, eyePl);
      ringB.add(tor, M(r * 0.5, r * 0.92, r * 0.92, r * 0.6), 0, undefined, 0, eyePl);
    }
    addMesh(irisB.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: e.iris, roughness: 0.35, metalness: 0.3 }), u), 2);
    addMesh(pupilB.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: '#030305', roughness: 0.1 }), u), 2);
    addMesh(ringB.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: e.ring ?? sp.color.side, roughness: 0.5 }), u), 2);
    addMesh(corneaB.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: '#ffffff', roughness: 0.02, transparent: true, opacity: 0.18, clearcoat: 1 }), u), 3);
  }

  // ── FINS, TAILS & SYMMETRICAL SIDE FINS ───────────────────────────────────
  const fs = sp.fins;
  const finMat = (o: { color?: string; tip?: string; alpha?: number; rays?: number; incise?: number; edge?: string; spots?: number; solid?: boolean }) => {
    const mem = o.alpha ?? fs.alpha;
    const solid = o.solid ?? (mem >= 0.99 && (o.rays ?? 14) === 0);
    return patchMaterial(
      new THREE.MeshPhysicalMaterial({
        color: '#ffffff',
        roughness: solid ? 0.46 : 0.4,
        side: THREE.DoubleSide,
        transparent: !solid,
        depthWrite: solid,
        clearcoat: 0.38,
      }),
      u,
      {
        rays: o.rays ?? 14,
        incise: o.incise ?? 0,
        membrane: mem,
        base: col(o.color ?? fs.color),
        tip: col(o.tip ?? o.color ?? fs.tip ?? fs.color),
        edge: col(o.edge ?? '#000'),
        edgeW: o.edge ? 0.12 : 0,
        spots: o.spots ?? 0,
        spotCol: col('#111'),
      },
    );
  };
  const finSpots = sp.color.layers?.some((l) => l.k === 'spots' && (l.h1 ?? 1) > 0.3 && l.n > 40) ? 3 : 0;

  // 1. Median fins (dorsals, anals, finlets) with curved rays & 3D hydrofoils for sharks/lobes/adipose
  const median = (f: MedianFin, side: 1 | -1) => {
    const isFoil = (f.foil ?? (f.shape === 'shark' || f.shape === 'lobe' || f.shape === 'adipose' ? 1 : 0)) > 0;
    const sw = (f.sweep ?? (f.shape === 'shark' ? 30 : f.shape === 'falcate' ? 34 : f.shape === 'spiny' ? 16 : 12)) * DEG;
    const bendBack = f.curve ?? (f.shape === 'falcate' || f.shape === 'veil' || f.shape === 'flag' ? 0.32 : f.shape === 'shark' ? 0.2 : 0.14);
    const nu = Math.max(18, Math.round((f.s1 - f.s0) * 150));
    const sampleMid = (uu: number, vv: number) => {
      const s = lerp(f.s0, f.s1, uu);
      const y = side > 0 ? model.top(s) * 0.92 : -model.bot(s) * 0.92;
      const B = model.local(s, y, 0);
      const fr = model.frame(s);
      const a = sw * (0.85 + 0.35 * uu);
      const dir = fr.N.clone().multiplyScalar(side * Math.cos(a)).addScaledVector(fr.T, Math.sin(a));
      const H = f.h * shapeH(f.shape, uu) * L;
      const vExt = vv * 1.05 - 0.05;
      const p = B.addScaledVector(dir, vExt * H).addScaledVector(fr.T, bendBack * vv * vv * H);
      return { p, H, s };
    };
    if (isFoil) {
      const foilAmt = f.foil ?? 1;
      const g = foilGrid(nu, 12, (uu, vv) => {
        const { p, s } = sampleMid(uu, vv);
        const tMax = Math.min(model.hw(s) * 0.55, f.h * 0.22) * foilAmt * L;
        return { p, n: new THREE.Vector3(0, 0, 1), t: tMax, flap: vv * 0.45, side: 0 };
      });
      const rays = f.shape === 'lobe' ? 10 : 0;
      addMesh(g, finMat({ ...f, rays, alpha: f.alpha ?? 1, solid: rays === 0, spots: f.spots ?? finSpots }), rays === 0 ? 2 : 4);
    } else {
      const g = grid(nu, 12, (uu, vv) => ({ p: sampleMid(uu, vv).p, flap: vv, side: 0 }));
      addMesh(g, finMat({ ...f, rays: f.rays ?? Math.round(nu / 2.1), spots: f.spots ?? finSpots }), 4);
    }
  };
  for (const f of fs.dorsal ?? []) median(f, 1);
  for (const f of fs.anal ?? []) median(f, -1);
  if (fs.finlets) {
    const fl = fs.finlets;
    for (let i = 0; i < fl.n; i++) {
      const s = lerp(fl.s0, fl.s1, i / Math.max(1, fl.n - 1));
      const w = ((fl.s1 - fl.s0) / fl.n) * 0.78;
      for (const sd of [1, -1] as const) median({ s0: s, s1: s + w, h: fl.h, shape: 'falcate', sweep: 38, rays: 3, color: fl.color, alpha: 0.96 }, sd);
    }
  }

  // 2. Caudal fin (curved leading margins, shark subterminal notch, 3D muscular hypural/vertebral base)
  buildCaudal(fs.caudal);
  function buildCaudal(cs: CaudalSpec) {
    if (cs.type === 'none' && cs.span <= 0) return;
    const len = cs.len ?? cs.span * 0.6;
    const up = cs.up ?? (cs.type === 'hetero' ? 0.68 : 0.5);
    const D: Record<string, { fork: number; p: number; lu: number; ld: number; tuft?: number }> = {
      forked: { fork: 0.38, p: 1.2, lu: 1, ld: 1 },
      lunate: { fork: 0.16, p: 0.82, lu: 1, ld: 1 },
      emarginate: { fork: 0.82, p: 1.65, lu: 1, ld: 1 },
      truncate: { fork: 0.96, p: 2.1, lu: 1, ld: 1 },
      rounded: { fork: 1, p: 2, lu: 0.52, ld: 0.52 },
      pointed: { fork: 1, p: 1, lu: 0.08, ld: 0.08 },
      hetero: { fork: 0.3, p: 1.12, lu: 1, ld: 0.48 },
      diphy: { fork: 0.8, p: 2, lu: 0.45, ld: 0.45, tuft: 0.45 },
      halfmoon: { fork: 1, p: 2.8, lu: 0.86, ld: 0.86 },
      lyre: { fork: 0.3, p: 2.6, lu: 1, ld: 1 },
      none: { fork: 1, p: 1, lu: 0.1, ld: 0.1 },
    };
    const d = D[cs.type];
    const lu = cs.lenUp ?? d.lu * len, ld = cs.lenDn ?? d.ld * len;
    const lc = (cs.fork ?? d.fork) * len;
    const pedTop = model.top(1), pedBot = model.bot(1);
    const isCartilaginous = sp.skin.kind === 'placoid' || cs.type === 'hetero';
    const isFoilTail = isCartilaginous || cs.type === 'lunate' || (cs.foil ?? 0) > 0;

    const sampleTail = (uu: number, vv: number) => {
      const yb = lerp(pedTop, -pedBot, uu) * 0.98;
      const sb = 0.972;
      let ySpanTop = cs.span * up, ySpanBot = -cs.span * (1 - up);
      let yE = lerp(ySpanTop, ySpanBot, uu) + sp.body.pedY;
      let xE: number;
      if (cs.type === 'rounded' || cs.type === 'halfmoon') {
        const th = (uu - 0.5) * Math.PI;
        const rLobe = cs.type === 'halfmoon' ? len : len * 1.85;
        xE = rLobe * Math.pow(Math.max(0, Math.cos(th)), cs.type === 'halfmoon' ? 0.35 : 0.55);
        yE = lerp(ySpanTop, ySpanBot, 0.5 - 0.5 * Math.cos(uu * Math.PI)) + sp.body.pedY;
      } else if (uu < up) {
        const t = 1 - uu / up; // 0 at fork, 1 at dorsal tip
        xE = lc + (lu - lc) * Math.pow(t, d.p);
        // authentic subterminal notch near the tip of shark heterocercal upper lobe
        if (cs.type === 'hetero') {
          const notch = Math.exp(-Math.pow((t - 0.82) / 0.065, 2));
          xE -= (lu - lc) * 0.17 * notch;
        }
      } else {
        const t = (uu - up) / Math.max(1e-4, 1 - up); // 0 at fork, 1 at ventral tip
        xE = lc + (ld - lc) * Math.pow(t, d.p);
      }
      if (d.tuft) xE += d.tuft * len * Math.exp(-Math.pow((uu - up) / 0.08, 2));

      const B = model.local(sb, yb, 0);
      const E = model.local(1 + xE, yE, 0);
      // organic convex curvature along dorsal & ventral outer margins (eliminates straight polygon edges)
      const vvX = Math.pow(vv, 1.12);
      const vvY = Math.pow(vv, 0.82);
      const p = new THREE.Vector3(lerp(B.x, E.x, vvX), lerp(B.y, E.y, vvY), 0);
      return { p, uu, vv };
    };

    if (isFoilTail) {
      const pedW = Math.max(0.008, model.hw(0.96)) * L;
      const g = foilGrid(44, 14, (uu, vv) => {
        // Note: in foilGrid u is chordwise, v is spanwise – for tail, vv is root->trailing edge (chord), uu is top->bottom
        const { p } = sampleTail(uu, vv);
        // muscular vertebral axis along upper lobe in heterocercal tails, symmetric hypural core in lunate tails
        const axisBias = cs.type === 'hetero' ? 0.35 + 0.85 * Math.exp(-Math.pow((uu - 0.12) / 0.22, 2)) : 0.7;
        const hypural = pedW * axisBias * (1 - 0.75 * Math.pow(vv, 0.65));
        return { p, n: new THREE.Vector3(0, 0, 1), t: hypural, flap: vv * 0.42, side: 0 };
      });
      const rays = isCartilaginous ? 0 : (cs.rays ?? 22);
      addMesh(g, finMat({ color: cs.color, tip: cs.tip, alpha: isCartilaginous ? 1 : (cs.alpha ?? 0.94), rays, solid: isCartilaginous, edge: cs.edge, spots: finSpots }), isCartilaginous ? 2 : 4);
    } else {
      const g = grid(44, 14, (uu, vv) => ({ p: sampleTail(uu, vv).p, flap: vv * 0.45, side: 0 }));
      addMesh(g, finMat({ color: cs.color, tip: cs.tip, alpha: cs.alpha, rays: cs.rays ?? 20, edge: cs.edge, spots: finSpots }), 4);
    }
  }

  // 3. Paired side fins (pectorals & pelvics) – singularity-free anatomical frame + EXACT z -> -z mirror symmetry
  const paired = (f: PairedFin) => {
    const at = model.attach(f.s, f.y, 1); // build right fin (+Z) first
    const fr = model.frame(f.s);
    const Z = new THREE.Vector3(0, 0, 1);
    const el = f.elev * DEG, sp_ = f.splay * DEG;
    const swExtra = ((f.sweep ?? (f.shape === 'shark' ? 24 : f.shape === 'falcate' ? 16 : 6)) * DEG) * Math.sin(sp_);

    // Primary span axis on the right side (+Z):
    const dir = fr.T.clone()
      .multiplyScalar(Math.cos(sp_) * Math.cos(el) + Math.sin(swExtra) * 0.48)
      .addScaledVector(fr.N, Math.sin(el))
      .addScaledVector(Z, Math.sin(sp_) * Math.cos(el * 0.42))
      .normalize();

    // Smooth leading-edge reference:
    //   small splay (teleost flank paddle) -> leading edge uu=0 is DORSAL (+N)
    //   large splay (shark/ray/flyingfish wing) -> leading edge uu=0 is ANTERIOR (-T, snoutward)
    const leadRef = fr.N.clone().multiplyScalar(Math.cos(sp_)).addScaledVector(fr.T, -Math.sin(sp_)).normalize();
    // Chord vector from leading edge (uu=0) to trailing edge (uu=1), orthogonal to dir:
    const bChord = leadRef.clone().negate().addScaledVector(dir, leadRef.dot(dir)).normalize();
    // Outward / upper fin normal on the right side (+Z -> +N as splay goes 0 -> 90°):
    const rawN = Z.clone().multiplyScalar(Math.cos(sp_)).addScaledVector(fr.N, Math.sin(sp_));
    const finN = rawN.addScaledVector(dir, -rawN.dot(dir)).addScaledVector(bChord, -rawN.dot(bChord));
    if (finN.lengthSq() < 1e-6) finN.crossVectors(dir, bChord);
    finN.normalize();
    if (finN.z < -0.05 && sp_ < 65 * DEG) finN.negate();

    const spread = (f.shape === 'fan' ? 52 : f.shape === 'veil' ? 32 : f.shape === 'round' ? 24 : 15) * DEG;
    const baseOrigin = at.p.clone().addScaledVector(at.n, -0.005 * L);
    const isFoil = (f.foil ?? (f.shape === 'shark' || f.shape === 'lobe' ? 1 : 0)) > 0;
    const sweepBack = f.shape === 'shark' || f.shape === 'falcate' ? 0.18 : 0.08;

    const sampleRight = (uu: number, vv: number) => {
      const a = (uu - 0.42) * spread;
      const H = f.len * shapeH(f.shape, uu) * L;
      const ray = dir.clone().multiplyScalar(Math.cos(a)).addScaledVector(bChord, Math.sin(a));
      const vIn = vv * 1.04 - 0.04;
      const p = baseOrigin.clone()
        .addScaledVector(bChord, (uu - 0.35) * f.base * L)
        .addScaledVector(ray, vIn * H)
        // natural backward sweep curve along rays + subtle hydrodynamic camber
        .addScaledVector(bChord, sweepBack * vv * vv * H)
        .addScaledVector(finN, Math.sin(vv * Math.PI * 0.65) * H * 0.045);
      return { p, H };
    };

    let rightG: THREE.BufferGeometry;
    const isSolid = isFoil && f.shape !== 'lobe' && (f.rays ?? (f.shape === 'shark' ? 0 : 12)) === 0;
    if (isFoil) {
      const foilAmt = f.foil ?? 1;
      const tRoot = Math.max(0.004, Math.min(f.base * 0.28, f.len * 0.085)) * foilAmt * L;
      rightG = foilGrid(22, 12, (uu, vv) => {
        const { p } = sampleRight(uu, vv);
        return { p, n: finN, t: tRoot, flap: vv, side: 1 };
      });
    } else {
      const checkFlip = new THREE.Vector3().crossVectors(bChord, dir).dot(finN) < 0;
      rightG = grid(20, 12, (uu, vv) => ({ p: sampleRight(uu, vv).p, flap: vv, side: 1 }), checkFlip);
    }
    const leftG = mirrorZ(rightG, -1);
    const rays = f.rays ?? (f.shape === 'shark' ? 0 : 12);
    const mat = finMat({ ...f, rays, alpha: isSolid ? 1 : f.alpha, solid: isSolid });
    addMesh(rightG, mat, isSolid ? 2 : 4, false);
    addMesh(leftG, mat, isSolid ? 2 : 4, true);
  };
  if (fs.pec) paired(fs.pec);
  if (fs.pel) paired(fs.pel);

  // ── ROSTRUM (bills, swords, saws, paddles, tubular snouts) ────────────────
  if (sp.rostrum) {
    const ro = sp.rostrum;
    const fr0 = model.frame(0.0);
    const fwd = fr0.T.clone().negate();
    const yb = ro.kind === 'tube' ? 0 : model.top(0.03) * 0.25;
    const start = model.local(0.04, yb, 0);
    const flat = ro.flat ?? 0;
    const g = grid(16, 40, (uu, vv) => {
      const t = vv * 1.1 - 0.1;
      let w = ro.r, h = ro.r;
      switch (ro.kind) {
        case 'bill': w = h = ro.r * Math.pow(1 - clamp(t), 0.75) + 0.002; break;
        case 'sword': w = ro.r * Math.pow(1 - clamp(t), 0.55) + 0.002; h = w * (1 - flat); break;
        case 'saw': w = ro.r * (1 - 0.35 * clamp(t)) + 0.002; h = w * (1 - flat); break;
        case 'paddle': w = ro.r * (0.55 + 1.5 * Math.pow(Math.sin(Math.PI * clamp(t) * 0.92), 0.7)); h = ro.r * 0.4 * (1 - 0.6 * clamp(t)); break;
        case 'tube': w = h = ro.r * (1 - 0.15 * clamp(t)); break;
      }
      const a = uu * Math.PI * 2;
      const tip = t > 0.985 ? smooth(1.1, 0.985, t) : 1;
      const p = start.clone().addScaledVector(fwd, t * ro.len * L)
        .addScaledVector(fr0.N, Math.cos(a) * h * L * tip)
        .add(new THREE.Vector3(0, 0, Math.sin(a) * w * L * tip));
      return { p };
    }, true);
    const rm = patchMaterial(new THREE.MeshPhysicalMaterial({ color: ro.color ?? sp.color.top, roughness: 0.45, clearcoat: 0.4, bumpMap: bump, bumpScale: 0.3 }), u);
    addMesh(g, rm, 1);
    if (ro.kind === 'saw') {
      const batch = new Batch();
      const cone = new THREE.ConeGeometry(0.3, 1, 5); cone.translate(0, 0.5, 0);
      const n = 24;
      for (let i = 0; i < n; i++) {
        const t = 0.05 + 0.92 * (i / (n - 1));
        const w = ro.r * (1 - 0.35 * t);
        for (const sd of [1, -1]) {
          const p = start.clone().addScaledVector(fwd, t * ro.len * L).add(new THREE.Vector3(0, 0, sd * w * L * 0.95));
          batch.add(cone, orient(p, new THREE.Vector3(0.25, 0, sd), 0.012 * L * 0.6, 0.03 * L, 0.012 * L * 0.6));
        }
      }
      addMesh(batch.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: '#efe8d8', roughness: 0.3 }), u), 2);
    }
    if (ro.kind === 'tube') {
      const tipP = start.clone().addScaledVector(fwd, ro.len * L);
      const disc = new THREE.CircleGeometry(ro.r * L * 0.65, 20);
      const batch = new Batch();
      batch.add(disc, new THREE.Matrix4().compose(tipP.addScaledVector(fwd, 0.002), new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0, 0, 1), fwd), new THREE.Vector3(1, 0.7, 1)));
      addMesh(batch.build(), patchMaterial(new THREE.MeshStandardMaterial({ color: '#1a0508', side: THREE.DoubleSide }), u), 2);
    }
  }

  // ── lamprey oral disc ─────────────────────────────────────────────────────
  if (m.kind === 'disc') {
    const fr0 = model.frame(0.02);
    const fwd = fr0.T.clone().negate();
    const dn = fr0.N.clone().negate();
    const face = fwd.clone().multiplyScalar(0.55).addScaledVector(dn, 0.85).normalize();
    const ctr = model.local(0.035, -model.bot(0.035) * 0.55, 0).addScaledVector(face, 0.01 * L);
    const r = model.bot(0.04) * 1.15 * L;
    const q = new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0, 0, 1), face);
    const b1 = new Batch(), b2 = new Batch(), b3 = new Batch();
    b1.add(new THREE.TorusGeometry(r, r * 0.22, 10, 40), new THREE.Matrix4().compose(ctr, q, new THREE.Vector3(1, 1, 1)));
    b2.add(new THREE.CircleGeometry(r, 40), new THREE.Matrix4().compose(ctr.clone().addScaledVector(face, -r * 0.1), q, new THREE.Vector3(1, 1, 1)));
    const cone = new THREE.ConeGeometry(0.3, 1, 5); cone.translate(0, 0.5, 0); cone.rotateX(Math.PI / 2);
    for (let ringI = 0; ringI < 3; ringI++)
      for (let i = 0; i < 14 + ringI * 6; i++) {
        const a = (i / (14 + ringI * 6)) * Math.PI * 2;
        const rr = r * (0.25 + ringI * 0.25);
        const p = new THREE.Vector3(Math.cos(a) * rr, Math.sin(a) * rr, 0).applyQuaternion(q).add(ctr);
        b3.add(cone, new THREE.Matrix4().compose(p, q, new THREE.Vector3(r * 0.12, r * 0.12, r * 0.2)));
      }
    addMesh(b1.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: sp.color.belly, roughness: 0.5 }), u), 2);
    addMesh(b2.build(), patchMaterial(new THREE.MeshStandardMaterial({ color: '#6a2c2c', side: THREE.DoubleSide }), u), 2);
    addMesh(b3.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: '#f0d890', roughness: 0.3 }), u), 3);
  }

  // ── BARBELS, LURES, CIRRI ─────────────────────────────────────────────────
  const tube = (pts: THREE.Vector3[], r0: number, r1: number, jawW: number, pl: [number, number, number, number]) => {
    const crv = new THREE.CatmullRomCurve3(pts);
    const nS = 24, nR = 7;
    const frames = crv.computeFrenetFrames(nS, false);
    return grid(nR, nS, (uu, vv) => {
      const i = Math.round(vv * nS);
      const p = crv.getPointAt(vv);
      const a = uu * Math.PI * 2;
      const r = lerp(r0, r1, vv);
      p.addScaledVector(frames.normals[i], Math.cos(a) * r).addScaledVector(frames.binormals[i], Math.sin(a) * r);
      return { p, jaw: jawW, flap: vv * 0.6, side: 0, plastic: pl };
    });
  };
  const barbelMat = patchMaterial(new THREE.MeshPhysicalMaterial({ color: sp.color.side, roughness: 0.5, clearcoat: 0.3 }), u);
  disposables.push(barbelMat);
  const bulbBatch = new Batch();
  let bulbColor = '#9ff';
  const barbelList = [...(sp.barbels ?? [])];
  if (sp.cirri) barbelList.push({ s: sp.eye.s, y: sp.eye.y + 0.25, len: 0.05, dir: [0.1, 1, 0.15], droop: -0.2, r: 0.004, pairs: true });
  for (const bb of barbelList) {
    for (const side of (bb.pairs === false ? [1] : [1, -1]) as (1 | -1)[]) {
      const at = model.attach(bb.s, bb.y, side);
      const fr = model.frame(bb.s);
      const d = fr.T.clone().multiplyScalar(-bb.dir[0]).addScaledVector(fr.N, bb.dir[1]).add(new THREE.Vector3(0, 0, side * bb.dir[2])).normalize();
      const pts: THREE.Vector3[] = [];
      for (let k = 0; k <= 5; k++) {
        const t = k / 5;
        pts.push(at.p.clone().addScaledVector(d, t * bb.len * L).addScaledVector(fr.N, -bb.droop * t * t * bb.len * L).addScaledVector(fr.T, 0.15 * t * t * bb.len * L));
      }
      if (bb.pairs === false) pts.forEach((p) => (p.z = 0));
      const yAbs = model.normY(bb.s, bb.y);
      const isLowerB = bb.s < rc.s + 0.01 && yAbs < model.mouthY(bb.s);
      const jw = isLowerB ? model.jawWeight(bb.s, 0.4) : seamJawUpper(bb.s, at.phi);
      const pl = model.plasticAt(bb.s, at.phi, isLowerB);
      addMesh(tube(pts, bb.r * L, bb.r * L * 0.25, jw, [pl[0] * 0.9, pl[1] * 0.5, pl[2] * 0.6, 0]), barbelMat, 2, true);
      if (bb.bulb) {
        bulbColor = bb.bulb;
        bulbBatch.add(new THREE.SphereGeometry(1, 16, 12), new THREE.Matrix4().compose(pts[5], new THREE.Quaternion(), new THREE.Vector3(1, 1, 1).multiplyScalar(bb.r * L * 2.6)), jw);
      }
      if (bb.pairs === false) break;
    }
  }
  if (!bulbBatch.empty) addMesh(bulbBatch.build(), patchMaterial(new THREE.MeshStandardMaterial({ color: bulbColor, emissive: bulbColor, emissiveIntensity: 1.6 }), u), 3);

  // ── SPIKES / HORNS / SCUTES / SPINES (symmetrically paired) ───────────────
  const spikeB = new Batch();
  const spikeCone = new THREE.ConeGeometry(0.35, 1, 6); spikeCone.translate(0, 0.5, 0);
  for (const k of sp.spikes ?? []) {
    for (const side of (k.mirror === false ? [1] : [1, -1]) as (1 | -1)[]) {
      const at = model.attach(k.s, k.y, side);
      if (k.mirror === false) at.p.z = 0;
      const fr = model.frame(k.s);
      const d = (k.dir[0] || k.dir[1] || k.dir[2])
        ? fr.T.clone().multiplyScalar(-k.dir[0]).addScaledVector(fr.N, k.dir[1]).add(new THREE.Vector3(0, 0, side * k.dir[2])).normalize()
        : at.n;
      const h = k.len * L, r = (k.r ?? 0.3) * h;
      spikeB.add(spikeCone, orient(at.p.clone().addScaledVector(at.n, -h * 0.1), d, r, h, r));
      if (k.mirror === false) break;
    }
  }
  if (sp.spines) {
    const pairsN = Math.ceil(sp.spines.n / 2);
    for (let i = 0; i < pairsN; i++) {
      const s = 0.08 + R() * 0.8;
      const yn = (R() * 2 - 1) * 0.94;
      const h = sp.spines.len * L * (0.65 + R() * 0.55);
      for (const side of [1, -1] as const) {
        const at = model.attach(s, yn, side);
        const fr = model.frame(s);
        const d = at.n.clone().multiplyScalar(0.55).addScaledVector(fr.T, 0.8).normalize();
        spikeB.add(spikeCone, orient(at.p, d, h * 0.16, h, h * 0.16));
      }
    }
  }
  if (!spikeB.empty) addMesh(spikeB.build(), patchMaterial(new THREE.MeshPhysicalMaterial({ color: sp.spines?.c ?? sp.color.side, roughness: 0.4, clearcoat: 0.3 }), u), 2);

  // ── FRONDS (seadragons) ───────────────────────────────────────────────────
  if (sp.fronds) {
    for (const [s, y, len] of sp.fronds.at) {
      const midline = Math.abs(y) > 0.92;
      const at = model.attach(s, y, 1);
      if (midline) at.p.z = 0;
      const out = midline ? model.frame(s).N.clone().multiplyScalar(Math.sign(y)) : at.n.clone();
      const nf = midline ? new THREE.Vector3(0, 0, 1) : new THREE.Vector3().crossVectors(out, new THREE.Vector3(1, 0, 0)).normalize();
      const b = new THREE.Vector3().crossVectors(nf, out).normalize();
      const frondMat = finMat({ color: sp.fronds.c, tip: sp.fronds.c, alpha: 0.9, rays: 3 });
      disposables.push(frondMat);
      for (let l = 0; l < 3; l++) {
        const a = (l - 1) * 0.52;
        const d = out.clone().multiplyScalar(Math.cos(a)).addScaledVector(b, Math.sin(a));
        const gR = grid(10, 8, (uu, vv) => {
          const H = len * L * (l === 1 ? 1 : 0.75);
          const w = 0.022 * L * Math.sin(Math.PI * vv) * (0.6 + 0.4 * Math.sin(vv * 9));
          const p = at.p.clone().addScaledVector(d, vv * H).addScaledVector(b, (uu - 0.5) * w * 2);
          return { p, flap: vv * 0.8, side: 0 };
        });
        addMesh(gR, frondMat, 4, true);
        if (!midline) addMesh(mirrorZ(gR, 0), frondMat, 4, true);
      }
    }
  }

  // ── animation constants & plan orientation ────────────────────────────────
  const swim = SWIM[sp.swim.mode] ?? SWIM.carangiform;
  const fwd0 = model.frame(0).T.clone().negate();
  u.uFwd.value.set(fwd0.x, fwd0.y).normalize();
  u.uHinge.value.copy(model.hingeXY());
  u.uMaxGape.value = Math.max(5, sp.mouth.open) * DEG;
  u.uProtrude.value = model.protrudeAmt();
  u.uGular.value = model.gularAmt();
  u.uL.value = L;
  u.uX0.value = model.local(0, 0, 0).x;
  u.uAmp.value = sp.body.bend ? 0 : swim.amp;
  u.uWave.value = swim.wave;
  u.uFreq.value = swim.freq;
  u.uA0.value = swim.a0;
  u.uPow.value = swim.pow;
  u.uFlap.value = swim.flap;
  u.uFinWave.value = swim.finWave;
  let wing = 0;
  if (sp.plan === 'ray') {
    wing = sp.id.includes('manta') ? 0.22 : 0.1;
    u.uSpan.value = (sp.body.disc?.span ?? 0.5) * L;
    u.uWingF.value = sp.id.includes('manta') ? 1.4 : 2.2;
  }
  u.uWing.value = wing;
  if (sp.plan === 'flatR') group.rotation.x = -Math.PI / 2;
  if (sp.plan === 'flatL') group.rotation.x = Math.PI / 2;

  const box = new THREE.Box3().setFromObject(group);
  const radius = box.getSize(new THREE.Vector3()).length() / 2;
  const ctr = box.getCenter(new THREE.Vector3());
  const localCtr = ctr.clone().applyQuaternion(group.quaternion.clone().invert());
  // preserve exact z = 0 symmetry plane for all bilateral fishes & rays!
  if (sp.plan === 'fish' || sp.plan === 'ray') localCtr.z = 0;
  group.children.forEach((ch) => ch.position.sub(localCtr));

  return {
    group, model, uniforms: u, swim, wing, radius,
    dispose: () => disposables.forEach((d) => d.dispose()),
  };
}
